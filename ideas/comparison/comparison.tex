\rowcolor{rowtwo}
{\tool{ChessStack}\par
\bullets{
  \item Dynamic website; free for one repertoire, with a fee for more.
  \item Since February 2026; 10 GitHub stars.
  \item Website: \link{https://chessstack.app}
  \item Source: \link{https://github.com/PvtTwinkle/Chessstack}
  \item License: GNU Affero General Public License, version 3 or later. \link{https://www.gnu.org/licenses/agpl-3.0.html}
}}
& \statusbad{\item Account required}
& \statusok{\item Rate-limited}
& \statusok{\item On server}
& \statusgood{\item Free Spaced Repetition Scheduler\item Models memory and schedules reviews just before they are likely to be forgotten}
& \bullets{
  \item Health score from 0 to 100: average of coverage (100 minus 5 for each gap against common opponent moves), learned positions, positions not due for review, and positions without repeated errors.
  \item Accuracy trend: share of correct answers in each of the last 14 sessions.
  \item Trouble spots: five most-failed positions, shown after at least three lapses.
  \item Streak: consecutive days with a completed practice session.
}
& \bullets{\item Opponent preparation from their games\item Game review against your repertoire\item Puzzles}
\\
